qemu-img create -f qcow2 -o preallocation=off template.qcow2 25G

virt-install \
-n template \
--osinfo=ubuntujammy \
--memory=4096 \
--vcpus=4,sockets=2,cores=2,threads=1 \
--cpu host \
--network type=direct,source=eth0,source_mode=bridge,model=virtio \
--disk='/var/lib/libvirt/images/template.qcow2',bus=virtio \
--location=/home/mnorouzi/Downloads/ubuntu-24.04.3-live-server-amd64.iso,kernel=casper/vmlinuz,initrd=casper/initrd \
--noautoconsole \
--graphics=vnc,password=password,listen=0.0.0.0

# to find the port:
virsh -c qemu:///system vncdisplay template


for connecting to the vm use vncviewer 127.0.0.1:5900

# snapshot
virsh snapshot-create-as template --name fresh-os

# revert to snapshot
virsh snapshot-revert template fresh-os

# delete snapshot
virsh snapshot-delete template fresh-os

# delete vm
virsh destroy template
virsh undefine template --remove-all-storage
